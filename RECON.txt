#!/usr/bin/env python3

import argparse
import json
import socket


def service_name(port):
    try:
        return socket.getservbyport(port, "tcp")
    except OSError:
        return "unknown"


def main():
    parser = argparse.ArgumentParser(
        description="GxNetRecon - reconocimiento TCP autorizado"
    )

    parser.add_argument("target")
    parser.add_argument("--ports", default="22,53,80,443,8080")
    parser.add_argument("--timeout", type=float, default=1.0)
    parser.add_argument(
        "-o",
        "--output",
        default="gxnetrecon_report.json"
    )

    args = parser.parse_args()

    try:
        ip = socket.gethostbyname(args.target)
    except socket.gaierror:
        raise SystemExit(
            "[!] No se pudo resolver el objetivo"
        )

    open_ports = []

    for value in args.ports.split(","):
        try:
            port = int(value.strip())
        except ValueError:
            print(f"[!] Puerto ignorado: {value}")
            continue

        if not 1 <= port <= 65535:
            continue

        with socket.socket(
            socket.AF_INET,
            socket.SOCK_STREAM
        ) as sock:

            sock.settimeout(args.timeout)

            if sock.connect_ex((ip, port)) == 0:
                service = service_name(port)

                open_ports.append({
                    "port": port,
                    "service": service
                })

                print(f"[OPEN] {port} ({service})")

    report = {
        "tool": "GxNetRecon",
        "target": args.target,
        "ip": ip,
        "open_ports": open_ports,
    }

    with open(
        args.output,
        "w",
        encoding="utf-8"
    ) as f:
        json.dump(
            report,
            f,
            indent=2,
            ensure_ascii=False
        )

    print(f"[+] Reporte: {args.output}")

    main()
